cmake_minimum_required(VERSION 3.28)
# Standalone CPU project: no CUDA compiler, checkpoint, SSH, or serving endpoint.
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(dsgb10_vision_host_tests LANGUAGES CXX)
  include(CTest)
endif()
get_filename_component(ENGINE_DIR "${CMAKE_CURRENT_LIST_DIR}/.." ABSOLUTE)
function(vision_host_test target source)
  add_executable(${target} "${ENGINE_DIR}/${source}")
  target_include_directories(${target} PRIVATE "${ENGINE_DIR}")
  target_compile_features(${target} PRIVATE cxx_std_20)
  if(MSVC)
    target_compile_options(${target} PRIVATE /UNDEBUG /EHsc)
  else()
    target_compile_options(${target} PRIVATE -UNDEBUG)
  endif()
  add_test(NAME ${target} COMMAND ${target})
endfunction()
if(BUILD_TESTING)
  vision_host_test(vision_prefix_test server/tests/vision_prefix_test.cpp)
  option(DSGB10_ENABLE_FILE_TESTS "Test image spool files in an isolated /tmp" OFF)
  if(DSGB10_ENABLE_FILE_TESTS AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
    vision_host_test(vision_budget_test server/tests/vision_budget_test.cpp)
  endif()
endif()
