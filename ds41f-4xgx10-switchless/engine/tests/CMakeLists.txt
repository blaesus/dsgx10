cmake_minimum_required(VERSION 3.28)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(dsgb10_probability_host_tests LANGUAGES CXX)
  include(CTest)
endif()
get_filename_component(DSGB10_ENGINE_DIR "${CMAKE_CURRENT_LIST_DIR}/.." ABSOLUTE)

if(BUILD_TESTING)
  add_executable(probability_host_test "${DSGB10_ENGINE_DIR}/model/tests/probability_host_test.cpp")
  target_include_directories(probability_host_test PRIVATE "${DSGB10_ENGINE_DIR}")
  target_compile_features(probability_host_test PRIVATE cxx_std_20)
  if(MSVC)
    target_compile_options(probability_host_test PRIVATE /UNDEBUG /EHsc /utf-8)
  else()
    target_compile_options(probability_host_test PRIVATE -UNDEBUG)
  endif()
  # Only header-defined host helpers are used; this does not link/initialize CUDA.
  find_path(DSGB10_CUDA_INCLUDE_DIR cuda_runtime.h
    HINTS "$ENV{CUDA_PATH}/include" ${CMAKE_CUDA_TOOLKIT_INCLUDE_DIRECTORIES}
          /usr/local/cuda/include /usr/local/cuda-13.0/include)
  if(DSGB10_CUDA_INCLUDE_DIR)
    target_include_directories(probability_host_test PRIVATE "${DSGB10_CUDA_INCLUDE_DIR}")
    target_compile_definitions(probability_host_test PRIVATE DSGB10_TEST_WITH_CUDA_HEADERS=1)
  else()
    message(STATUS "CUDA headers absent: probability_host_test runs core helpers; batch verdict section is skipped")
  endif()
  add_test(NAME probability_host_test COMMAND probability_host_test)
  find_package(Python3 COMPONENTS Interpreter QUIET)
  if(Python3_Interpreter_FOUND)
    add_test(NAME distribution_oracle
      COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/distribution_oracle.py")
  endif()
endif()

# Explicit build only: CTest never launches GPU/rank/network operations.
option(DSGB10_BUILD_GPU_PROBE "Build the opt-in four-rank probability sampler probe" OFF)
if(DSGB10_BUILD_GPU_PROBE)
  if(NOT TARGET dsmodel)
    message(FATAL_ERROR "GPU probe requires the full engine CUDA project")
  endif()
  add_executable(probabilistic_sample_probe probabilistic_sample_probe.cu)
  target_include_directories(probabilistic_sample_probe PRIVATE "${CMAKE_CURRENT_LIST_DIR}")
  target_link_libraries(probabilistic_sample_probe PRIVATE dsmodel)
  target_compile_options(probabilistic_sample_probe PRIVATE -UNDEBUG)
endif()
